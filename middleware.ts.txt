import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

export async function middleware(request: NextRequest) {
  // On crée la réponse de base
  let response = NextResponse.next({ request });

  // On initialise Supabase avec la gestion des cookies
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        get(name) {
          return request.cookies.get(name)?.value;
        },
        set(name, value, options) {
          response.cookies.set({ name, value, ...options });
        },
        remove(name, options) {
          response.cookies.set({ name, value: '', ...options });
        },
      },
    }
  );

  // On récupère l'utilisateur
  const { data: { user } } = await supabase.auth.getUser();

  // 1. Si on est sur /admin et que l'utilisateur n'est PAS connecté -> Login
  if (request.nextUrl.pathname.startsWith('/admin') && !user) {
    return NextResponse.redirect(new URL('/login', request.url));
  }

  // 2. Si on est sur /admin et que l'utilisateur n'est PAS admin -> Dashboard
  if (request.nextUrl.pathname.startsWith('/admin') && user) {
    const { data: profile } = await supabase
      .from('profiles')
      .select('role')
      .eq('id', user.id)
      .single();

    if (profile?.role !== 'admin') {
      return NextResponse.redirect(new URL('/dashboard', request.url));
    }
  }

  return response;
}

// ⚠️ TRÈS IMPORTANT : On applique le middleware UNIQUEMENT sur /admin
// Cela évite qu'il ne tourne sur /login et crée une boucle infinie
export const config = {
  matcher: ['/admin/:path*'],
};
